  1213→function computePnlForTrade(t: SeedTrade): number {
  1214→  if (t.status === "open") return 0;
  1215→  const dir = t.direction === "long" ? 1 : -1;
  1216→  return (t.exitPrice - t.entryPrice) * t.positionSize * dir - t.fees;
  1217→}
  1218→
  1219→function computeRRForTrade(t: SeedTrade): number | null {
  1220→  if (t.direction === "long") {
  1221→    const risk = t.entryPrice - t.stopLoss;
  1222→    const reward = t.takeProfit - t.entryPrice;
  1223→    if (risk === 0) return null;
  1224→    return reward / risk;
  1225→  } else {
  1226→    const risk = t.stopLoss - t.entryPrice;
  1227→    const reward = t.entryPrice - t.takeProfit;
  1228→    if (risk === 0) return null;
  1229→    return reward / risk;
  1230→  }
  1231→}
